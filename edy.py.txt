import pandas as pd
import numpy as np
from pathlib import Path

DATA_PATH = Path(r'C:\Users\gabri\.copilot\session-state\7cbd1390-7a8b-4cbe-b727-b8288e90c785\files\database.tsv')


def section(title: str):
    """Print a section title."""
    print(f"\n=== {title} ===")


def find_col(df: pd.DataFrame, name: str):
    """Return actual column name matching `name` case-insensitively, or None."""
    for c in df.columns:
        if c.lower() == name.lower():
            return c
    return None


def load_data(path: Path) -> pd.DataFrame:
    df = pd.read_csv(path, sep='\t', skiprows=2)
    print(f"Loaded {len(df)} rows and {len(df.columns)} columns from {path.name}")
    return df


def show_overview(df: pd.DataFrame):
    section("Columns (first 15)")
    for i, col in enumerate(df.columns[:15], start=1):
        print(f"{i}. {col}")
    remaining = len(df.columns) - 15
    if remaining > 0:
        print(f"... and {remaining} more columns")

    section("Sample rows")
    cols_of_interest = ['Category', 'Country', 'product', 'author']
    existing = []
    for c in cols_of_interest:
        found = find_col(df, c)
        if found:
            existing.append(found)
    if existing:
        print(df[existing].head(3).to_string())
    else:
        print("No requested sample columns found.")


def show_missing_data(df: pd.DataFrame, top_n: int = 15):
    section("Missing data")
    missing = df.isnull().sum()
    missing_pct = (missing / len(df) * 100).round(2)
    summary = pd.DataFrame({'missing': missing, 'percent': missing_pct})
    summary = summary[summary['missing'] > 0].sort_values('missing', ascending=False)
    if summary.empty:
        print("No columns have missing data.")
        return
    print(f"Top {top_n} columns with the most missing values:")
    print(summary.head(top_n).to_string())


def show_value_counts(df: pd.DataFrame, column: str, label: str = None):
    label = label or column
    section(f"Distribution {label}")
    col = find_col(df, column)
    if col is None:
        print(f"(column '{column}' does not exist in the dataset)")
        return
    counts = df[col].value_counts(dropna=False)
    print(counts.to_string())
    print(f"\nTotal unique values: {len(counts)}")


def show_stanford_entries(df: pd.DataFrame):
    section("Entries related to Stanford")
    col = find_col(df, 'author')
    if col is None:
        print("(column 'author' does not exist)")
        return
    mask = df[col].astype(str).str.contains('Stanford', case=False, na=False)
    count = mask.sum()
    print(f"Found {count} records with 'Stanford' in the author field")
    if count == 0:
        return
    cols = [find_col(df, c) for c in ['product', 'Category', 'Country', 'author', 'title']]
    cols = [c for c in cols if c is not None]
    for _, row in df[mask][cols].iterrows():
        print()
        for c in cols:
            print(f" {c}: {row[c]}")


def show_usa_entries(df: pd.DataFrame):
    section("Entries from USA / California")
    col = find_col(df, 'Country')
    if col is None:
        print("(column 'Country' does not exist)")
        return
    mask = df[col].astype(str).str.contains('USA|United States|California|\\bCA\\b', case=False, na=False)
    print(f"Records from USA: {mask.sum()}")


def show_sample_size_stats(df: pd.DataFrame):
    section("Sample size statistics")
    col = find_col(df, 'Sample size (n)')
    if col is None:
        print(f"(column 'Sample size (n)' does not exist)")
        return
    sizes = pd.to_numeric(df[col].astype(str).str.extract(r'([0-9]+)')[0], errors='coerce')
    print(f"Mean: {sizes.mean():.1f}")
    print(f"Median: {sizes.median():.1f}")
    print(f"Min: {sizes.min()}, Max: {sizes.max()}")


def show_publication_years(df: pd.DataFrame):
    section("Publication years")
    col = find_col(df, 'published')
    if col is None:
        print(f"(column 'published' does not exist)")
        return
    years = pd.to_numeric(df[col], errors='coerce')
    years = years.dropna().astype(int)
    if years.empty:
        print("No valid year data found.")
        return
    year_counts = years.value_counts().sort_index()
    print(f"Range: {int(years.min())} - {int(years.max())}")
    print("Last 10 years with records:")
    print(year_counts.tail(10).to_string())


def main():
    df = load_data(DATA_PATH)
    show_overview(df)
    show_missing_data(df)
    show_value_counts(df, 'Category')
    show_value_counts(df, 'Country')
    show_stanford_entries(df)
    show_usa_entries(df)
    show_sample_size_stats(df)
    show_publication_years(df)


if __name__ == '__main__':
    main()
